cmake_minimum_required(VERSION 3.24)
project(StereoForge VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(STEREOFORGE_RHI "Build the pinned Slang RHI Vulkan native stereo runtime" OFF)
set(STEREOFORGE_RHI_PROVIDER "standalone" CACHE STRING "RHI provider: standalone production or slangpy research")
set_property(CACHE STEREOFORGE_RHI_PROVIDER PROPERTY STRINGS standalone slangpy)
option(STEREOFORGE_CAPTURE "Build ZED Open Capture adapter" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(stereoforge_contracts INTERFACE)
target_include_directories(stereoforge_contracts INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(BUILD_TESTING)
  add_executable(native_contracts tests/native_contracts.cpp)
  target_link_libraries(native_contracts PRIVATE stereoforge_contracts)
  add_test(NAME native_contracts COMMAND native_contracts)
endif()
if(STEREOFORGE_RHI)
  if(STEREOFORGE_RHI_PROVIDER STREQUAL "standalone")
    include(cmake/StandaloneRHI.cmake)
  elseif(STEREOFORGE_RHI_PROVIDER STREQUAL "slangpy")
  find_package(Python REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python_EXECUTABLE}" -c "import importlib.metadata, pathlib, slangpy; assert importlib.metadata.version('slangpy') == '0.43.1'; print(pathlib.Path(slangpy.__file__).parent)"
    OUTPUT_VARIABLE SLANGPY_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  # The wheel omits these public helper headers. Fetch only the matching compiler
  # revision, checking bytes; all RHI headers/binaries come from the pinned wheel.
  set(SLANG_HELPERS "${CMAKE_CURRENT_BINARY_DIR}/slang-helpers")
  file(MAKE_DIRECTORY "${SLANG_HELPERS}")
  file(DOWNLOAD "https://raw.githubusercontent.com/shader-slang/slang/v2026.12/include/slang-com-ptr.h" "${SLANG_HELPERS}/slang-com-ptr.h"
    EXPECTED_HASH SHA256=0bb72f8f4135ab01ab5c6c27f8da478bfbcf1a6132cf33d153b2357303664905 TLS_VERIFY ON)
  file(DOWNLOAD "https://raw.githubusercontent.com/shader-slang/slang/v2026.12/include/slang-com-helper.h" "${SLANG_HELPERS}/slang-com-helper.h"
    EXPECTED_HASH SHA256=8c4936e5bad6e61222cde9ae266dc5931cdcd5989d01495cef6dff2aef9b9276 TLS_VERIFY ON)
  add_library(stereoforge_rhi_provider INTERFACE)
  target_link_libraries(stereoforge_rhi_provider INTERFACE "${SLANGPY_ROOT}/libsgl.so" "${SLANGPY_ROOT}/libslang-compiler.so.0.2026.12")
  target_include_directories(stereoforge_rhi_provider SYSTEM INTERFACE "${SLANGPY_ROOT}/include" "${SLANG_HELPERS}")
  find_path(SF_VULKAN_HEADERS vulkan/vulkan_core.h REQUIRED)
  target_include_directories(stereoforge_rhi_provider SYSTEM INTERFACE "${SF_VULKAN_HEADERS}")
  set(SF_RHI_RUNTIME_LIBRARY_DIR "${SLANGPY_ROOT}")
  else()
    message(FATAL_ERROR "Unknown STEREOFORGE_RHI_PROVIDER: ${STEREOFORGE_RHI_PROVIDER}")
  endif()
  find_package(Threads REQUIRED)
  add_library(stereoforge_runtime src/runtime.cpp src/geometry.cpp src/native_device.cpp)
  target_link_libraries(stereoforge_runtime PUBLIC stereoforge_contracts stereoforge_rhi_provider PRIVATE
    ${CMAKE_DL_LIBS} Threads::Threads)
  set_target_properties(stereoforge_runtime PROPERTIES BUILD_RPATH "${SF_RHI_RUNTIME_LIBRARY_DIR}")
  add_executable(native_stereo apps/native_stereo.cpp)
  target_link_libraries(native_stereo PRIVATE stereoforge_runtime)
  set_target_properties(native_stereo PROPERTIES BUILD_RPATH "${SF_RHI_RUNTIME_LIBRARY_DIR}")
  target_compile_definitions(native_stereo PRIVATE STEREOFORGE_SHADER_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/shaders/stereo")
  add_executable(native_geometry apps/native_geometry.cpp)
  target_link_libraries(native_geometry PRIVATE stereoforge_runtime)
  set_target_properties(native_geometry PROPERTIES BUILD_RPATH "${SF_RHI_RUNTIME_LIBRARY_DIR}")
  target_compile_definitions(native_geometry PRIVATE STEREOFORGE_SHADER_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/apps/native_view_benchmark.cpp")
    add_executable(native_view_benchmark apps/native_view_benchmark.cpp)
    target_link_libraries(native_view_benchmark PRIVATE stereoforge_runtime)
    set_target_properties(native_view_benchmark PROPERTIES BUILD_RPATH "${SF_RHI_RUNTIME_LIBRARY_DIR}")
    target_compile_definitions(native_view_benchmark PRIVATE STEREOFORGE_SHADER_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
  endif()
  if(BUILD_TESTING)
    add_executable(native_gpu_handoff tests/native_gpu_handoff.cpp)
    target_link_libraries(native_gpu_handoff PRIVATE stereoforge_runtime)
    set_target_properties(native_gpu_handoff PROPERTIES BUILD_RPATH "${SF_RHI_RUNTIME_LIBRARY_DIR}")
    target_compile_definitions(native_gpu_handoff PRIVATE STEREOFORGE_SHADER_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/shaders" STEREOFORGE_STEREO_SHADER_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/shaders/stereo")
    add_test(NAME native_gpu_handoff COMMAND native_gpu_handoff)
    set_tests_properties(native_gpu_handoff PROPERTIES LABELS "vulkan;hardware")
    add_executable(native_geometry_contracts tests/native_geometry.cpp)
    target_link_libraries(native_geometry_contracts PRIVATE stereoforge_runtime)
    set_target_properties(native_geometry_contracts PROPERTIES BUILD_RPATH "${SF_RHI_RUNTIME_LIBRARY_DIR}")
    target_compile_definitions(native_geometry_contracts PRIVATE STEREOFORGE_SHADER_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
    # The pinned RHI otherwise silently runs without Khronos validation. Force
    # an empty explicit-layer search path and verify our required-layer guard.
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/empty-vulkan-layers")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/require-validation-rejection.cmake" CONTENT [=[
execute_process(COMMAND "${EXECUTABLE}" --validation RESULT_VARIABLE result OUTPUT_VARIABLE output ERROR_VARIABLE error)
if(result STREQUAL "0" OR NOT error MATCHES "VK_LAYER_KHRONOS_validation is unavailable")
  message(FATAL_ERROR "Required validation rejection failed: ${result} ${output} ${error}")
endif()
]=])
    add_test(NAME native_validation_required COMMAND "${CMAKE_COMMAND}" "-DEXECUTABLE=$<TARGET_FILE:native_stereo>" -P "${CMAKE_CURRENT_BINARY_DIR}/require-validation-rejection.cmake")
    set_tests_properties(native_validation_required PROPERTIES ENVIRONMENT "VK_LAYER_PATH=${CMAKE_CURRENT_BINARY_DIR}/empty-vulkan-layers" LABELS "vulkan;hardware")
    add_test(NAME native_vulkan_geometry COMMAND native_geometry_contracts)
    set_tests_properties(native_vulkan_geometry PROPERTIES LABELS "vulkan;hardware")
    add_test(NAME native_vulkan_stereo COMMAND native_stereo --self-test)
    set_tests_properties(native_vulkan_stereo PROPERTIES LABELS "vulkan;hardware")
  endif()
endif()
if(STEREOFORGE_CAPTURE)
  include(cmake/Capture.cmake)
endif()
